# Django Forms — Form-Level Custom Validation

## 1. Create the Form

### `students/forms.py`

```python
from django import forms
class StudentForm(forms.Form):
    name = forms.CharField(
        required=True,
        label="Student Name",
        initial="John",
        help_text="Enter your full name.",
        error_messages={"required": "Please enter your name."},
        widget=forms.TextInput()
    )

    dob = forms.DateField(
        required=True,
        label="Date of Birth",
        help_text="Enter your date of birth.",
        widget=forms.DateInput(attrs={"type": "date"})
    )

    gender = forms.ChoiceField(
        required=True,
        label="Gender",
        choices=[
            ("Male", "Male"),
            ("Female", "Female"),
            ("Others", "Others"),
        ],
        help_text="Select your gender."
    )

# Field-Level Custom Validation
    def clean_name(self):
        name = self.cleaned_data["name"]
        if len(name) < 3:
            raise forms.ValidationError(
                "Name must be at least 3 characters long."
            )
        return name
    
# Form-Level Custom Validation
    def clean(self):

        cleaned_data = super().clean()

        name = cleaned_data.get("name")
        dob = cleaned_data.get("dob")

        if name and dob:
            if name.lower() == "admin" and dob.year > 2008:
                raise forms.ValidationError(
                    "Admin is not a valid student and age must be greater than 18."
                )

        return cleaned_data
```
