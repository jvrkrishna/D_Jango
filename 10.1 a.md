This code:

\`\`\`python

\`\`\`sql

we can use:

\`\`\`python

\`\`\`

```html
<!DOCTYPE html>
```

# 🛠 Step 19: Create App-Level CRUD Views

Open:

```text
Sample_Project/myapp/views.py
```

Keep the existing `myapp_view()` and `rk_posts_view()` functions and add CRUD views.

```python
from django.shortcuts import render, redirect, get_object_or_404
from .models import Post


# Show all posts
def myapp_view(request):
    posts = Post.objects.all()

    return render(
        request,
        'myapp/myapp_home.html',
        {
            'posts': posts
        }
    )


# Show only posts by author "RK"
def rk_posts_view(request):
    posts = Post.objects.filter(author="RK")

    return render(
        request,
        'myapp/myapp_home.html',
        {
            'posts': posts
        }
    )


# CREATE
def create_post(request):
    if request.method == 'POST':
        title = request.POST.get('title')
        content = request.POST.get('content')
        author = request.POST.get('author')

        Post.objects.create(
            title=title,
            content=content,
            author=author
        )

        return redirect('all_posts')

    return render(request, 'myapp/create_post.html')


# READ
def read_posts(request):
    posts = Post.objects.all()

    return render(
        request,
        'myapp/read_posts.html',
        {
            'posts': posts
        }
    )


# UPDATE
def update_posts(request):
    posts = Post.objects.all()

    return render(
        request,
        'myapp/update_posts.html',
        {
            'posts': posts
        }
    )


def update_post(request, post_id):
    post = get_object_or_404(Post, id=post_id)

    if request.method == 'POST':
        post.title = request.POST.get('title')
        post.content = request.POST.get('content')
        post.author = request.POST.get('author')
        post.save()

        return redirect('update_posts')

    return render(
        request,
        'myapp/update_post.html',
        {
            'post': post
        }
    )


# DELETE
def delete_posts(request):
    posts = Post.objects.all()

    return render(
        request,
        'myapp/delete_posts.html',
        {
            'posts': posts
        }
    )


def delete_post(request, post_id):
    post = get_object_or_404(Post, id=post_id)

    if request.method == 'POST':
        post.delete()
        return redirect('delete_posts')

    return render(
        request,
        'myapp/delete_post.html',
        {
            'post': post
        }
    )
```

# 🧠 Final Concept

The project now demonstrates both Django Admin CRUD and application-level CRUD.

```text
                    Browser
                       ↓
                 App-level URLs
                       ↓
                    Views
                       ↓
                 Post Model
                       ↓
                    Database
                       ↓
                  Templates
                       ↓
                   Browser
```

At the same time, the common `base.html` provides:

```text
Header
  ↓
Admin Panel Menu
  ↓
Create | Read | Update | Delete
  ↓
Page Content
  ↓
Footer
```

This keeps the common layout in one place while each app-level page supplies its own `{% block content %}`.
